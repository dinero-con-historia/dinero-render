def sh(cmd):
    subprocess.run(cmd, check=True)

def load(path):
    sh(['ffmpeg', '-y', '-v', 'error', '-i', path, '-ac', '1', '-ar', str(SR), '/tmp/_l.wav'])
    sr, x = wf.read('/tmp/_l.wav')
    return x.astype(np.float32) / 32768

async def edge(text, voice, rate, pitch, path):
    await edge_tts.Communicate(text, voice, rate=rate, pitch=pitch).save(path)

def edge_phrase(text, voice, rate, pitch):
    asyncio.run(edge(text, voice, rate, pitch, '/tmp/_p.mp3'))
    return load('/tmp/_p.mp3')

def breath(n):
    x = np.random.randn(n).astype(np.float32)
    x = np.convolve(x, np.ones(24) / 24, 'same')
    return x * np.hanning(n) * 0.05

def assemble(chunks, pauses):
    out = []
    for i, c in enumerate(chunks):
        c = c.copy(); f = int(0.006 * SR)
        c[:f] *= np.linspace(0, 1, f); c[-f:] *= np.linspace(1, 0, f)
        out.append(c)
        p = int(pauses[i] * SR)
        sil = np.zeros(p, np.float32)
        if pauses[i] >= 0.8:
            b = breath(int(0.22 * SR)); sil[-len(b):] += b
        out.append(sil)
    return np.concatenate(out)

def finish(x, path):
    wf.write('/tmp/_raw.wav', SR, (np.clip(x, -1, 1) * 32767).astype(np.int16))
    sh(['ffmpeg', '-y', '-v', 'error', '-i', '/tmp/_raw.wav', '-af', FX, '-ar', str(SR), '-ac', '1', path])

def edge_voice(voice, base_rate, base_pitch, idx):
    random.seed(idx)
    chunks = []
    for j, t in enumerate(['Voz %s.' % NUM[idx]] + PH):
        r = base_rate + random.choice([-3, -1, 0, 2]); p = base_pitch + random.choice([-1, 0, 1])
        chunks.append(edge_phrase(t, voice, '%+d%%' % r, '%+dHz' % p))
    return assemble(chunks, [0.7] + PA)

def kokoro_voice(pipe, voice, speed, phrasal, idx):
    def say(t):
        a = []
        for _, _, au in pipe(t, voice=voice, speed=speed):
            a.append(au.numpy() if hasattr(au, 'numpy') else np.array(au))
        wf.write('/tmp/_k.wav', 24000, (np.clip(np.concatenate(a), -1, 1) * 32767).astype(np.int16))
        return load('/tmp/_k.wav')
    if phrasal:
        return assemble([say('Voz %s.' % NUM[idx])] + [say(t) for t in PH], [0.7] + PA)
    full = say('Voz %s.' % NUM[idx]); full2 = say(' '.join(PH))
    return np.concatenate([full, np.zeros(int(0.7 * SR), np.float32), full2])
